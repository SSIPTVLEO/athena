## 2.1 — Criar projeto em supabase.com (grátis)
## 2.2 — No SQL Editor, execute:

-- Tabela de comandos
CREATE TABLE IF NOT EXISTS commands (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    agent_id    TEXT NOT NULL,
    command     TEXT NOT NULL,
    action      TEXT,
    params      JSONB,
    priority    TEXT DEFAULT 'low',
    source      TEXT DEFAULT 'dashboard',
    confidence  FLOAT DEFAULT 1.0,
    created_at  TIMESTAMPTZ DEFAULT NOW()
);

-- Tabela de resultados
CREATE TABLE IF NOT EXISTS results (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    command_id  UUID REFERENCES commands(id),
    agent_id    TEXT NOT NULL,
    success     BOOLEAN,
    output      TEXT,
    duration    FLOAT,
    timestamp   TIMESTAMPTZ DEFAULT NOW()
);

-- Tabela de agentes registrados
CREATE TABLE IF NOT EXISTS agents (
    id           TEXT PRIMARY KEY,
    hostname     TEXT,
    platform     TEXT,
    capabilities JSONB,
    last_seen    TIMESTAMPTZ DEFAULT NOW(),
    online       BOOLEAN DEFAULT FALSE
);

-- Tabela de padrões aprendidos (sincronizado com agente)
CREATE TABLE IF NOT EXISTS learned_patterns (
    id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    agent_id      TEXT,
    pattern       TEXT UNIQUE,
    action        TEXT,
    params        JSONB,
    success_count INT DEFAULT 0,
    fail_count    INT DEFAULT 0,
    confidence    FLOAT DEFAULT 0.5,
    last_used     TIMESTAMPTZ DEFAULT NOW()
);

-- Tabela de alertas
CREATE TABLE IF NOT EXISTS alerts (
    id        UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    level     TEXT NOT NULL,
    title     TEXT NOT NULL,
    message   TEXT,
    agent_id  TEXT,
    seen      BOOLEAN DEFAULT FALSE,
    timestamp TIMESTAMPTZ DEFAULT NOW()
);

-- Índices para performance
CREATE INDEX IF NOT EXISTS idx_commands_agent ON commands(agent_id);
CREATE INDEX IF NOT EXISTS idx_results_command ON results(command_id);
CREATE INDEX IF NOT EXISTS idx_alerts_seen ON alerts(seen);

# ARQUITETURA DE APRENDIZADO
# ═══════════════════════════════════════════════════════
#
# O agente aprende com CADA execução:
# - Armazena padrão do comando + ação + resultado
# - Calcula taxa de sucesso por padrão
# - Quando confiança > 80%, executa sem chamar IA
# - Isso economiza tokens e executa mais rápido
#
# Banco local: athena_local.db (SQLite no PC do agente)
# Banco nuvem: Supabase (sincronizado)
#
# Nível de alertas:
#   info     → informativo, cor azul
#   warning  → atenção, cor amarela
#   high     → urgente, cor laranja
#   critical → crítico, cor vermelha (pisca)
